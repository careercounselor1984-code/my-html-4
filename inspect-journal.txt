  async function clgRestoreAllData(file){
    if(!file) return;
    if(file.size > 20 * 1024 * 1024){
      alert('백업 파일이 너무 큽니다. 올바른 통합본 전체 백업 파일인지 확인해주세요.');
      return;
    }
    let data;
    try{ data = JSON.parse(await file.text()); }
    catch(e){ alert('JSON 형식의 올바른 백업 파일이 아닙니다.'); return; }
    if(!clgValidateBackup(data)){
      alert('상담일지 통합본에서 만든 전체 백업 파일이 아닙니다.');
      return;
    }
    const count = Object.keys(data.localStorage).length + Object.keys(data.sessionStorage || {}).length;
    const ok = confirm(`전체 백업 ${count}개 항목을 복원할까요?\n\n현재 이 통합본에 저장된 전화번호부·메모·카톡 문구·상담일지·메뉴 설정은 백업 파일의 내용으로 교체됩니다.`);
    if(!ok) return;

    const previousLocal = clgCollectStorage(window.localStorage);
    const previousSession = clgCollectStorage(window.sessionStorage);
    try{
      window.localStorage.clear();
      Object.entries(data.localStorage).forEach(([key,value])=>{
        if(typeof value === 'string') window.localStorage.setItem(key,value);
      });
      window.sessionStorage.clear();
      Object.entries(data.sessionStorage || {}).forEach(([key,value])=>{
        if(typeof value === 'string') window.sessionStorage.setItem(key,value);
      });
    }catch(e){
      // 복원 도중 저장용량 오류 등이 발생하면 기존 데이터로 되돌림
      try{
        window.localStorage.clear();
        Object.entries(previousLocal).forEach(([key,value])=>window.localStorage.setItem(key,value));
        window.sessionStorage.clear();
        Object.entries(previousSession).forEach(([key,value])=>window.sessionStorage.setItem(key,value));
      }catch(_){}
      alert('전체 복원 중 오류가 발생하여 기존 저장내용으로 되돌렸습니다.');
      return;
    }
    alert(`전체 복원이 완료되었습니다.\n${count}개 저장 항목을 적용한 뒤 통합본을 다시 엽니다.`);
    window.location.reload();
  }
